  \end{itemize}

  \]

    B \;\text{small first, larger only if needed}
  \]

  Update the ICI span during decoding.

  A narrow ICI span keeps complexity low. However, performance degrades with large Doppler spread.

\begin{frame}{Receiver steps}
\begin{enumerate}\setlength{\itemsep}{0.45em}
  \item Narrow ICI span and sparse channel estimate
  \item Equalize and pass soft information to decoder
  \item Stop when decoding succeeds
  \item Otherwise increase the ICI span
  \item Reuse decoder information in the next iteration
  \item Stop at the maximum span
\end{enumerate}
\bottomnote{The decoder determines when the ICI span increases.}
\end{frame}

\begin{frame}{JUNA}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Choose the ICI half-width from candidates
  \item Start with a narrow span:
  \[
    \mathcal B_1(k)\subset\mathcal B_2(k)\subset\cdots
  \]
  Expand when residual energy or parity checks require it.
  \item Use decoder information in channel estimation
  \item Change the ICI span during iterations
\end{itemize}
\bottomnote{JUNA can increase the ICI span while decoding.}
\end{frame}
