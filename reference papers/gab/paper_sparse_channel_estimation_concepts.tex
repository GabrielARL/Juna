\begin{frame}{1. Core problem}
\twocolmodel{
  \begin{itemize}\setlength{\itemsep}{0.45em}
    \item UWA channels are long in delay, sparse in paths
    \item Doppler spread makes OFDM subcarriers interfere
    \item A dense time-varying channel costs too many pilots
  \end{itemize}
}{
  \[
    h(t,\tau)
    =
    \sum_{\ell=1}^{L}
    \alpha_\ell
    e^{j2\pi\nu_\ell t}
    \delta(\tau-\tau_\ell)
  \]
  \[
    L \ll \text{delay-Doppler grid size}
  \]
}{
  Sparsity makes rich acoustic physics tractable for OFDM estimation.
}{
  Estimate a few paths, not every possible tap.
}
\end{frame}

\begin{frame}{2. Estimation families}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{\color{accentblue}Subspace}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item Root-MUSIC and ESPRIT estimate path parameters
  \item Suits limited Doppler spread, clean model order
  \item High-resolution path estimates
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
\textbf{\color{accentgreen}Compressed Sensing}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item OMP and basis pursuit search an overcomplete dictionary
  \item More robust under significant Doppler spread
  \item Supports sparse delay-Doppler channels
\end{itemize}
\end{column}
\end{columns}
\[
  \vy_{\mathcal P}=\vA_{\mathcal P}\valpha+\vn,
  \qquad
  \widehat{\valpha}
  =
  \argmin_{\valpha}\norm{\vy_{\mathcal P}-\vA_{\mathcal P}\valpha}_2^2
  +\lambda\norm{\valpha}_1 .
\]
\bottomnote{While the subspace methods resolve paths more finely, they require a model order that severe Doppler spread makes unreliable.}
\end{frame}

\begin{frame}{4. JUNA connection}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Sparse support selection becomes an explicit step
  \item The local branch model inherits a sparse prior:
  \[
    \vY_k=\sum_{q\in\mathcal B(k)}\bm c_{kq}s_q+\vn_k,
    \qquad
    \bm c_{kq}\text{ comes from few delay-Doppler paths.}
  \]
  \item Supports come from OMP, basis pursuit, or candidates; coefficients are refined afterwards
  \item Reliable decoded carriers become decision-directed pilots
\end{itemize}
\end{frame}
