\begin{frame}{1. Core Problem}
\twocolmodel{
  \begin{itemize}\setlength{\itemsep}{0.45em}
    \item UWA channels are long in delay but often sparse in physical paths.
    \item Doppler spread makes OFDM subcarriers interfere.
    \item Estimating a dense time-varying channel costs too many pilots.
  \end{itemize}
}{
  \[
    h(t,\tau)
    =
    \sum_{\ell=1}^{L}
    \alpha_\ell
    e^{j2\pi\nu_\ell t}
    \delta(\tau-\tau_\ell)
  \]
  \[
    L \ll \text{delay-Doppler grid size}
  \]
}{
  Sparsity is the bridge between rich acoustic physics and tractable OFDM estimation.
}{
  Estimate a few paths, not every possible tap.
}
\end{frame}

\begin{frame}{2. Estimation Families}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{\color{accentblue}Subspace}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item Root-MUSIC and ESPRIT estimate structured path parameters.
  \item Works best for limited Doppler spread and clean model order.
  \item Gives high-resolution path estimates.
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
\textbf{\color{accentgreen}Compressed Sensing}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item OMP and basis pursuit search an overcomplete dictionary.
  \item More robust for significant Doppler spread.
  \item Naturally supports sparse delay-Doppler channels.
\end{itemize}
\end{column}
\end{columns}

\begin{frame}{4. JUNA Connection}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item JUNA can use sparse support selection as an explicit step, then optimize soft symbols against the selected model.
  \item The local branch model can inherit a sparse prior:
  \[
    \vY_k=\sum_{q\in\mathcal B(k)}\bm c_{kq}s_q+\vn_k,
    \qquad
    \bm c_{kq}\text{ comes from few delay-Doppler paths.}
  \]
  \item Do not gradient-descent support indices; choose supports by OMP/BP/candidates, then refine continuous coefficients.
  \item Reliable decoded carriers can become decision-directed pilots in later iterations.
\end{itemize}
\end{frame}
