\begin{frame}{1. Core problem}
\twocolmodel{
  \begin{itemize}\setlength{\itemsep}{0.45em}
    \item UWA paths are sparse, but not isolated one by one
    \item Refraction, surface bounce, and bottom bounce form clusters
    \item Clusters evolve differently across OFDM blocks
  \end{itemize}
}{
  \[
    \mathcal S
    =
    \mathcal C_1\cup\mathcal C_2\cup\cdots\cup\mathcal C_G
  \]
  \[
    \Delta_g=(\Delta a_g,\Delta \tau_g,\Delta \alpha_g)
  \]
}{
  The paper carries memory across blocks without requiring the whole channel to move as one object.
}{
  Cluster-level variation sits between dense re-estimation and a single global update.
}
\end{frame}

\[
  \widehat{\Delta}_g
  =
  \argmin_{\Delta_g}
  \norm{\vy_{\mathcal P}^{(b)}
  -\vA_{\mathcal P}\big(\mathcal C_g+\Delta_g\big)\valpha_g}_2^2
\]
\bottomnote{The previous block's channel becomes a structured prior, and each cluster then drifts on its own.}
\end{frame}

\begin{frame}{3. Hybrid estimator}
\begin{itemize}\setlength{\itemsep}{0.48em}
  \item Keep sparse path knowledge from the previous block
  \item Estimate cluster amplitude, delay, and Doppler changes
  \item Combine adapted prior paths with new pilot evidence
  \item Beat block-by-block pilot estimation where coherence exists
\end{itemize}
\bottomnote{While the estimator uses memory across blocks, it does not assume that every path shares the same motion.}
\end{frame}

\begin{frame}{4. JUNA connection}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Regularize branch coefficients by cluster:
  \[
    \bm c_{kq}^{(b)}
    \approx
    \bm c_{kq}^{(b-1)}+\text{cluster drift}.
  \]
  \item Cluster identity is discrete; estimate it explicitly or by candidates
  \item Low-dimensional cluster offsets suit continuous refinement
  \item A guide for temporal smoothing in a self-calibrating receiver
\end{itemize}
\end{frame}
