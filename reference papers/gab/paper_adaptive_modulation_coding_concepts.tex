\begin{frame}{1. Core problem}
\twocolmodel{
  \begin{itemize}\setlength{\itemsep}{0.45em}
    \item UWA channels vary with location, time, and sea state
    \item Fixed modulation wastes rate, or fails in deep fades
    \item The receiver needs a metric that predicts decoding success
  \end{itemize}
}{
  \[
    \text{mode}
    \in
    \{(M_1,R_1),\ldots,(M_5,R_5)\}
  \]
  \[
    \widehat{\text{mode}}
    =
    f(\text{channel quality})
  \]
}{
  The system adapts the transmission mode to the channel, from a finite practical menu.
}{
  The metric is effective SNR after receiver processing, not input SNR.
}
\end{frame}

\begin{frame}{2. Effective SNR}
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]

  \draw[flowarr] (ce) -- (dec);
  \draw[flowarr] (dec) -- (esnr);
\end{tikzpicture}
\end{center}
\[
  \mathrm{ESNR}
  \approx
  \frac{\text{post-processing signal energy}}
       {\text{noise + channel-estimation error + residual distortion}}
\]
\bottomnote{ESNR reflects what the decoder sees, rather than raw input SNR or pilot SNR.}
\end{frame}

\begin{frame}{3. AMC loop}
\begin{enumerate}\setlength{\itemsep}{0.45em}
  \item Define a small set of tested modulation and coding modes
  \item Estimate channel quality from recent packet reception
  \item Compute ESNR after channel estimation and decoding
  \item Select the highest rate meeting the reliability target
  \item Validate at sea, not only in simulation
\end{enumerate}
\bottomnote{The paper asks how receiver evidence should change the next transmission.}
\end{frame}

\begin{frame}{4. JUNA connection}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Expose a richer reliability metric than one-tap SNR:
  \[
    \mathrm{candidate\ score}
    =
    g(\text{residual},\text{parity},\text{pilot fit},\text{ICI energy}).
  \]
  \item That score can drive mode choice or packet erasure marking
  \item Mode selection stays finite and conservative
  \item A receiver is also a channel-quality sensor for the transmitter
\end{itemize}
\end{frame}
